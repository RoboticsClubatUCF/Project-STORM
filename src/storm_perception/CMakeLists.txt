cmake_minimum_required(VERSION 3.16)
project(storm_perception)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
ament_python_install_package(${PROJECT_NAME})
install(PROGRAMS scripts/ground_filter scripts/vlp_scan scripts/sensor_limits DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(segmentation test/test_ground.py)
  ament_add_pytest_test(sensor_projection test/test_scan.py)
  ament_add_pytest_test(range_gate test/test_limits.py)
endif()
ament_package()
