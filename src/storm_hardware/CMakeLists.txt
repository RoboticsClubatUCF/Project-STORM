cmake_minimum_required(VERSION 3.16)
project(storm_hardware LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
option(STORM_BUILD_CAN_DRIVER "Build the driver; requires pinned sparkcan library" OFF)
if(STORM_BUILD_CAN_DRIVER)
  find_package(rclcpp REQUIRED)
  find_package(rclcpp_lifecycle REQUIRED)
  find_package(hardware_interface REQUIRED)
  find_package(pluginlib REQUIRED)
  find_package(sparkcan CONFIG REQUIRED)
  add_library(storm_spark_can SHARED src/spark_can_hardware.cpp)
  target_compile_features(storm_spark_can PUBLIC cxx_std_17)
  target_include_directories(storm_spark_can PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
  target_link_libraries(storm_spark_can sparkcan::sparkcan)
  ament_target_dependencies(storm_spark_can rclcpp rclcpp_lifecycle hardware_interface pluginlib)
  pluginlib_export_plugin_description_file(hardware_interface storm_hardware.xml)
  install(TARGETS storm_spark_can EXPORT export_storm_spark_can LIBRARY DESTINATION lib)
  install(DIRECTORY include/ DESTINATION include)
  ament_export_targets(export_storm_spark_can HAS_LIBRARY_TARGET)
  ament_export_dependencies(rclcpp rclcpp_lifecycle hardware_interface pluginlib sparkcan)
  if(BUILD_TESTING)
    find_package(ament_cmake_gtest REQUIRED)
    ament_add_gtest(test_hardware_init test/test_hardware_init.cpp)
    target_link_libraries(test_hardware_init storm_spark_can)
  endif()
endif()
ament_package()
